# configuration file /etc/nginx/conf.d/madina.conf:
server {
    listen 80;
    server_name _;

    # Proxy backend API. Incoming path /backend/... is rewritten to /... for Laravel.
    location ^~ /backend/ {
        rewrite ^/backend/(.*)$ /$1 break;
        proxy_pass http://127.0.0.1:8000/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
        proxy_set_header X-Forwarded-Prefix /backend;
        proxy_redirect off;
        proxy_buffering off;
    }

    # Proxy frontend SPA under /frontend/
    location ^~ /frontend/ {
        rewrite ^/frontend/(.*)$ /$1 break;
        proxy_pass http://127.0.0.1:3000/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
        proxy_set_header X-Forwarded-Prefix /frontend;
        proxy_redirect off;
        proxy_buffering off;
    }

    # Serve SPA at /
    location / {
        proxy_pass http://127.0.0.1:3000/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
        proxy_set_header X-Forwarded-Prefix /;
        proxy_redirect off;
        proxy_buffering off;
    }

    # Healthcheck
    location = /healthz {
        add_header Content-Type text/plain;
        return 200 'OK';
    }
}
